// Probe: can Direct2D render on the GPU into a GDI DC on this host (hidden desktop included)?
#include <jni.h>
#include <windows.h>
#include <d2d1.h>
#include <dwrite.h>

extern "C" JNIEXPORT jlongArray JNICALL Java_D2DProbe_draw (JNIEnv *env, jclass, jlong hdc, jint w, jint h, jint type) {
	jlong out [6] = {0, 0, 0, 0, 0, 0};
	LARGE_INTEGER f, t0, t1, t2;
	QueryPerformanceFrequency (&f);
	QueryPerformanceCounter (&t0);
	ID2D1Factory *factory = NULL;
	HRESULT hr = D2D1CreateFactory (D2D1_FACTORY_TYPE_SINGLE_THREADED, __uuidof (ID2D1Factory), NULL, (void **) &factory);
	out [0] = hr;
	if (FAILED (hr)) goto done;
	{
		D2D1_RENDER_TARGET_PROPERTIES props = D2D1::RenderTargetProperties (
			type == 1 ? D2D1_RENDER_TARGET_TYPE_HARDWARE : type == 2 ? D2D1_RENDER_TARGET_TYPE_SOFTWARE : D2D1_RENDER_TARGET_TYPE_DEFAULT,
			D2D1::PixelFormat (DXGI_FORMAT_B8G8R8A8_UNORM, D2D1_ALPHA_MODE_IGNORE), 0, 0, D2D1_RENDER_TARGET_USAGE_NONE, D2D1_FEATURE_LEVEL_DEFAULT);
		ID2D1DCRenderTarget *target = NULL;
		hr = factory->CreateDCRenderTarget (&props, &target);
		out [1] = hr;
		if (FAILED (hr)) goto release;
		RECT rect = {0, 0, w, h};
		hr = target->BindDC ((HDC) hdc, &rect);
		out [2] = hr;
		IDWriteFactory *dwrite = NULL;
		DWriteCreateFactory (DWRITE_FACTORY_TYPE_SHARED, __uuidof (IDWriteFactory), (IUnknown **) &dwrite);
		IDWriteTextFormat *format = NULL;
		if (dwrite) dwrite->CreateTextFormat (L"Segoe UI", NULL, DWRITE_FONT_WEIGHT_NORMAL, DWRITE_FONT_STYLE_NORMAL, DWRITE_FONT_STRETCH_NORMAL, 12.0f, L"en-us", &format);
		ID2D1SolidColorBrush *red = NULL, *black = NULL;
		target->CreateSolidColorBrush (D2D1::ColorF (1, 0, 0, 1), &red);
		target->CreateSolidColorBrush (D2D1::ColorF (0, 0, 0, 1), &black);
		QueryPerformanceCounter (&t1);
		for (int frame = 0; frame < 100; frame++) {
			target->BeginDraw ();
			target->Clear (D2D1::ColorF (1, 1, 1, 1));
			for (int row = 0; row < 20; row++) {
				if (row == 3) target->FillRectangle (D2D1::RectF (0, row * 18.0f, (float) w, row * 18.0f + 18), red);
				if (format) target->DrawText (L"root 12 / 7 label", 17, format, D2D1::RectF (20, row * 18.0f, (float) w, row * 18.0f + 18), black);
			}
			hr = target->EndDraw ();
			if (FAILED (hr)) break;
		}
		QueryPerformanceCounter (&t2);
		out [3] = hr;
		out [4] = (t1.QuadPart - t0.QuadPart) * 1000000 / f.QuadPart;
		out [5] = (t2.QuadPart - t1.QuadPart) * 1000000 / f.QuadPart / 100;
		if (red) red->Release ();
		if (black) black->Release ();
		if (format) format->Release ();
		if (dwrite) dwrite->Release ();
		target->Release ();
	}
release:
	factory->Release ();
done:
	jlongArray result = env->NewLongArray (6);
	env->SetLongArrayRegion (result, 0, 6, out);
	return result;
}
