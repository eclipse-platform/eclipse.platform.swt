// Manual win32 probe (default package: runs on the official SWT jar too). Run it on the hidden desktop.
// Classpath needs the compiled test classes (EXTRA_CP=<out>/tests).

import java.lang.reflect.*;
import java.util.Random;

import org.eclipse.swt.SWT;
import org.eclipse.swt.internal.win32.OS;
import org.eclipse.swt.widgets.*;

/**
 * Replays the readonly Combo differential's seeds 0 .. lastSeed with the test's own op generator
 * (a new combo per seed, as the test runs), then creates the next seed's fresh combo and prints its
 * face hash with the UI cue state (WM_QUERYUISTATE: 1 hidefocus, 2 hideaccel) of the shell and of
 * each fresh combo, per seed. The Combo differential diverges at seed 8 step 0 on the face only, so
 * the state a seed leaves behind for the next combo is what this probe measures.
 * Arguments: outDir [lastSeed=7] [traceSeed=-1]; with traceSeed, the shell's cue state is printed after
 * every step of that seed with the op that ran.
 */
public class ComboSeedFaceProbe {
	public static void main (String [] args) throws Exception {
		int lastSeed = args.length > 1 ? Integer.parseInt (args [1]) : 7;
		int traceSeed = args.length > 2 ? Integer.parseInt (args [2]) : -1;
		Class<?> test = Class.forName ("org.eclipse.swt.tests.junit.Test_org_eclipse_swt_widgets_Combo_ViewportDifferential");
		Class<?> subjectType = Class.forName ("org.eclipse.swt.tests.junit.Test_org_eclipse_swt_widgets_Combo_ViewportDifferential$Subject");
		Class<?> harness = Class.forName ("org.eclipse.swt.tests.junit.ViewportDifferential");
		Method randomOp = test.getDeclaredMethod ("randomOp", Random.class, StringBuilder.class);
		Method render = harness.getMethod ("render", Control.class);
		Method hash = harness.getMethod ("hash", org.eclipse.swt.graphics.ImageData.class);
		randomOp.setAccessible (true);
		Display display = new Display ();
		Shell shell = new Shell (display);
		shell.setBounds (20, 20, 330, 420);
		Constructor<?> constructor = subjectType.getDeclaredConstructor (Shell.class, int.class);
		constructor.setAccessible (true);
		Object subject = constructor.newInstance (shell, SWT.DROP_DOWN | SWT.READ_ONLY);
		Method reset = subjectType.getDeclaredMethod ("reset");
		reset.setAccessible (true);
		Field comboField = subjectType.getDeclaredField ("combo");
		comboField.setAccessible (true);
		for (int s = 0; s <= lastSeed + 1; s++) {
			reset.invoke (subject);
			Combo combo = (Combo) comboField.get (subject);
			while (display.readAndDispatch ()) {}
			System.out.printf ("%s seed %d fresh face %s | uistate shell %d combo %d focus %s%n",
				System.getProperty ("swt.widget.mode", "official"), s,
				hash.invoke (null, render.invoke (null, combo)),
				OS.SendMessage (shell.handle, 0x0129 /* WM_QUERYUISTATE */, 0, 0),
				OS.SendMessage (combo.handle, 0x0129, 0, 0),
				display.getFocusControl () == combo ? "combo" : String.valueOf (display.getFocusControl ()));
			if (s > lastSeed) break;
			Random random = new Random (0xC0B0L + s);
			StringBuilder log = new StringBuilder ();
			for (int step = 0; step < 150; step++) {
				Object op = randomOp.invoke (null, random, log);
				Method apply = op.getClass ().getMethods () [0];
				for (Method m : op.getClass ().getMethods ()) if (m.getName ().equals ("apply")) apply = m;
				apply.setAccessible (true);
				apply.invoke (op, subject);
				while (display.readAndDispatch ()) {}
				if (s == traceSeed) {
					String [] names = log.toString ().split ("\n");
					System.out.printf ("%s seed %d step %3d %-22s uistate shell %d dropped %b%n", System.getProperty ("swt.widget.mode", "official"),
						s, step, names [names.length - 1], OS.SendMessage (shell.handle, 0x0129, 0, 0), combo.getListVisible ());
				}
			}
			if (s == traceSeed) {
				/* the test's reset () step by step: where between two seeds the cue state changes */
				String mode = System.getProperty ("swt.widget.mode", "official");
				System.out.printf ("%s reset: before dispose uistate %d dropped %b%n", mode, OS.SendMessage (shell.handle, 0x0129, 0, 0), combo.getListVisible ());
				combo.dispose ();
				System.out.printf ("%s reset: after dispose uistate %d%n", mode, OS.SendMessage (shell.handle, 0x0129, 0, 0));
				Combo next = new Combo (shell, SWT.DROP_DOWN | SWT.READ_ONLY);
				next.setBounds (5, 5, 300, 30);
				System.out.printf ("%s reset: after create uistate %d%n", mode, OS.SendMessage (shell.handle, 0x0129, 0, 0));
				next.setVisibleItemCount (20);
				for (int i = 0; i < 300; i++) next.add ("row " + i);
				System.out.printf ("%s reset: after adds uistate %d%n", mode, OS.SendMessage (shell.handle, 0x0129, 0, 0));
				next.setFocus ();
				System.out.printf ("%s reset: after setFocus uistate %d%n", mode, OS.SendMessage (shell.handle, 0x0129, 0, 0));
				while (display.readAndDispatch ()) {}
				System.out.printf ("%s reset: after settle uistate %d face %s%n", mode, OS.SendMessage (shell.handle, 0x0129, 0, 0),
					hash.invoke (null, render.invoke (null, next)));
				break;
			}
		}
		display.dispose ();
	}
}
