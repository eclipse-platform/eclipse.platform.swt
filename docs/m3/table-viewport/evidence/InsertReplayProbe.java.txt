package org.eclipse.swt.tests.junit;

import java.lang.reflect.Field;
import java.util.Random;

import org.eclipse.swt.SWT;
import org.eclipse.swt.widgets.*;

/*
 * Replays variant 8 (MULTI | VIRTUAL, reentrancy script) of the Table differential up to the step whose insert throws
 * on the official jar, then performs that insert from its own listener with the stack trace and the keyed store.
 * args: seed, step, event (Selection | DefaultSelection | SetData), trigger (key name or getText row fraction)
 */
public class InsertReplayProbe {
	static int setDataDepth;

	public static void main (String [] args) throws Exception {
		int variant = 8, seed = Integer.parseInt (args [0]), stop = Integer.parseInt (args [1]);
		int type = args [2].equals ("Selection") ? SWT.Selection : args [2].equals ("DefaultSelection") ? SWT.DefaultSelection : SWT.SetData;
		Display display = Display.getDefault ();
		Test_org_eclipse_swt_widgets_Table_ViewportDifferential.Scene scene = new Test_org_eclipse_swt_widgets_Table_ViewportDifferential.Scene (display, Test_org_eclipse_swt_widgets_Table_ViewportDifferential.VARIANTS [variant]);
		scene.reset ();
		Random random = new Random (0x7AB1E5EEDL + 31L * variant + seed);
		StringBuilder log = new StringBuilder ();
		for (int step = 0; step < stop; step++) {
			scene.events.setLength (0);
			TableViewportScripts.next (Test_org_eclipse_swt_widgets_Table_ViewportDifferential.REENTRANT, random, log).apply (scene);
		}
		Table table = scene.table;
		System.out.println ("SWT " + SWT.getVersion () + " rows " + table.getItemCount () + " top " + table.getTopIndex () + " keys " + keys (table));
		boolean [] done = {false};
		table.addListener (SWT.SetData, e -> {
			if (done [0] || type == SWT.SetData) return;
			if (insideCreate ()) System.out.println ("SetData for row " + e.index + " inside createItem, keys " + keys (table));
		});
		table.addListener (type, e -> {
			if (done [0]) return;
			done [0] = true;
			System.out.println ("inside " + args [2] + " for " + (e.item == null ? "-" : ((TableItem) e.item).getText ()) + ", keys " + keys (table));
			try {
				new TableItem (table, SWT.NONE, 0).setText ("inner");
				System.out.println ("insert ok, keys " + keys (table));
			} catch (RuntimeException x) {
				StringBuilder frames = new StringBuilder ();
				for (StackTraceElement frame : x.getStackTrace ()) {
					if (frames.length () > 700) break;
					frames.append (frame.getClassName ().replace ("org.eclipse.swt.widgets.", "")).append ('.').append (frame.getMethodName ()).append (':').append (frame.getLineNumber ()).append (' ');
				}
				System.out.println ("insert threw " + x + " | keys " + keys (table) + " | " + frames);
			}
		});
		if (args [3].equals ("getText")) {
			table.getItem (Integer.parseInt (args [4])).getText ();
		} else {
			TableViewportScripts.key (scene, TableViewportScripts.vk (args [3]), false, false);
		}
		TableViewportScripts.quiesce (scene);
		System.out.println ("done " + done [0]);
		display.dispose ();
	}

	static boolean insideCreate () {
		for (StackTraceElement frame : Thread.currentThread ().getStackTrace ()) {
			if (frame.getMethodName ().equals ("createItem")) return true;
		}
		return false;
	}

	static String keys (Table table) {
		try {
			Field count = Table.class.getDeclaredField ("keyCount"), keys = Table.class.getDeclaredField ("keys");
			count.setAccessible (true);
			keys.setAccessible (true);
			int [] array = (int []) keys.get (table);
			return count.getInt (table) + "/" + (array == null ? "dense" : array.length);
		} catch (ReflectiveOperationException e) {
			return "?";
		}
	}
}
